# Interval for the robot scheduler, in seconds
cogwork_tick (Scheduler interval) float 0.1 0.05 1

# Delay in seconds for a robot's action (minimum 0.1)
cogwork_action_delay (Action delay) float 0.2 0.1 10

# Delay in seconds for a robot's movement (minimum 0.1)
cogwork_move_delay (Movement delay) float 0.5 0.1 10

# Allow the chat command on public robots
cogwork_public_chat (Allow public chat) bool true

# Let the placed item's own on_place handler run (seeds, torches, ...)
cogwork_use_on_place (Use item on_place) bool true

# Start robots that were running when the server stopped
cogwork_resume_on_start (Resume on start) bool false

# Statements run per scheduler tick before a program yields
cogwork_step_budget (Step budget) int 200 10 5000

# Largest radius, in blocks, a robot can watch for entity and item events
cogwork_event_max_radius (Event radius limit) int 6 1 32

# Seconds between scans at the largest radius (smaller radii scan faster, never below 0.5)
cogwork_event_max_interval (Event scan interval at the largest radius) float 3 0.5 30

# Chat messages a robot may send (public or private) or receive per player ...
cogwork_chat_limit (Chat messages per window) int 5 1 100

# ... within this many seconds
cogwork_chat_window (Chat window) float 10 1 600

# Most block cells a robot looks at when it plans a path (longer, more winding routes fail)
cogwork_path_max_nodes (Pathfinding search limit) int 6000 100 100000

# Seconds between a travelling robot's re-planning of its path (terrain and moving targets change)
cogwork_path_replan (Path re-plan interval) float 5 0.5 120

# How far, in blocks, a robot looks for the entity it was told to walk to
cogwork_path_entity_radius (Pathfinding entity search radius) float 48 1 200

# Robots that are not running fall to the floor
cogwork_gravity (Gravity) bool true

# A robot in water floats in it (it can be flooded, like a waterlogged block). When off it sinks to the bottom
cogwork_robots_can_float (Robots can float) bool true

# How far a stopped robot looks for a floor to fall to, in blocks (none found: it stays where it is)
cogwork_fall_range (Fall range) int 128 1 1000

# Power a Self-Generator puts into one power cell per interval
cogwork_generator_amount (Self-Generator power) float 10 1 1000

# Seconds between a Self-Generator's charges (never under 5)
cogwork_generator_interval (Self-Generator interval) float 5 5 600

# Thruster particles under robots that fly or hover
cogwork_particles (Thruster particles) bool true

# Let players enter a robot (sneak + punch): a room with a viewscreen, chest, controls
cogwork_rooms (Rooms inside robots) bool true

# Picture of the viewscreen inside a robot, in pixels (the screen is 8 x 5 blocks)
cogwork_view_width (Viewscreen width) int 96 16 256
cogwork_view_height (Viewscreen height) int 60 10 160

# How far, in blocks, the viewscreen sees (loaded terrain only)
cogwork_view_range (Viewscreen range) float 96 16 256

# Seconds between two pictures of the viewscreen
cogwork_view_interval (Viewscreen interval) float 1 0.5 60
